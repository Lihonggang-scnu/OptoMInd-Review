162:     def classify(exc: Exception) -> TransportError:
163:         if isinstance(exc, urllib.error.HTTPError):
164:             code = int(exc.code)
165:             if code == 429:
166:                 return TransportError("rate_limit", "provider 429")
167:             if code in (401, 403):
168:                 return TransportError("auth_error", f"provider {code}")
169:             if code < 500:
170:                 # billing/auth/param rejections: provider refused before inference
171:                 return TransportError("auth_error", f"provider {code} rejected request")
172:             return TransportError("transport_error", f"provider {code}")
173:         name = type(exc).__name__
174:         if name in ("RemoteDisconnected", "IncompleteRead", "TimeoutError",
175:                     "socket.timeout", "http.client.RemoteDisconnected"):
176:             # the request may have been processed: budget must stay frozen
177:             return TransportError("transport_error",
178:                                   f"connection lost mid-flight ({name})",
179:                                   "may_have_processed=true")
180:         if isinstance(exc, urllib.error.URLError):
181:             # never reached the provider: safe to release the reservation
182:             return TransportError("transport_error", f"connection failed ({name})",
183:                                   "may_have_processed=false")
184:         return TransportError("transport_error", name)
