142:     def _settle_one(ledger, reservation_id, receipt):
143:         row = next((r for r in ledger.as_dict()["reservations"] if r["reservation_id"] == reservation_id), None)
144:         if row is None:
145:             raise ValueError("dual_unknown_reservation")
146:         desired = "uncertain" if receipt["uncertain"] else "settled"
147:         if row["status"] in {"settled", "uncertain"}:
148:             actual = receipt["actual_cny"]

150:                 row.get("actual_cny") is not None and actual is not None and
151:                 abs(float(row["actual_cny"]) - float(actual)) < 1e-9)
152:             if row["status"] != desired or not same_amount:

158:     def _finish_settlement(self, record):
159:         self._check(record)
160:         errors = []
161:         for name, ledger in (("project", self.project), ("total", self.total)):
162:             reservation_id = record.get(name + "_reservation_id")
163:             if reservation_id:
164:                 try:
165:                     self._settle_one(ledger, reservation_id, record["settlement"])
166:                 except Exception as exc:
167:                     errors.append({"ledger": name, "error_type": type(exc).__name__,
168:                                    "reason": "settlement_failed_hold_retained"})
169:         record.update(state="settlement_incomplete" if errors else
170:                       "uncertain" if record["settlement"]["uncertain"] else "settled",
171:                       settlement_errors=errors)

176:     def settle(self, reservation_id, actual_cny, *, uncertain=False, telemetry=None):
177:         record = read_json(self._path(reservation_id))
178:         self._check(record)
179:         receipt = {"actual_cny": actual_cny, "uncertain": bool(uncertain),
180:                    "telemetry": dict(telemetry or {})}
181:         if record.get("settlement") is not None and record["settlement"] != receipt:
182:             raise QwenTransportError("dual_conflicting_settlement", transient=False)
183:         record.update(settlement=receipt, state="settling")
184:         write_json(self._path(reservation_id), record)  # Save evidence before either side settles.
