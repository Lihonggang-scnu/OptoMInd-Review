542:     def _actual_micro(usage: Dict[str, Any]) -> int:
543:         """Actual cost from provider usage fields; falls back to explicit cny."""
544:         if "actual_cost_micro" in usage:
545:             return int(usage["actual_cost_micro"])
546:         pt = int(usage.get("prompt_tokens") or 0)
547:         ct = int(usage.get("completion_tokens") or 0)
548:         rates = usage.get("rates") or {}
549:         micro = (pt * float(rates.get("input_cny_per_million", 0))
550:                  + ct * float(rates.get("output_cny_per_million", 0))) * MICRO / 1_000_000
551:         return int(round(micro))
